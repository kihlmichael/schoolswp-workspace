foreach ($items as $item) {
    $id = (int)$item['id'];
    if (isset($item['alt'])) { update_post_meta($id, '_wp_attachment_image_alt', $item['alt']); }
    $upd = ['ID' => $id];
    if (isset($item['caption'])) $upd['post_excerpt'] = $item['caption'];
    if (isset($item['description'])) $upd['post_content'] = $item['description'];
    if (count($upd) > 1) {
        $r = wp_update_post($upd, true);
        if (is_wp_error($r)) { $errors[] = $id . ': ' . $r->get_error_message(); }
        else { $updated++; }
    } elseif (isset($item['alt'])) { $updated++; }
