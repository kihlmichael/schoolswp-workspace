// Expand each pair to 3 variants: curly apostrophe, HTML entity &rsquo;, straight quote '
$replacements = [];
foreach ($pairs as $pair) {
  $from = $pair[0];
  $to = $pair[1];
  $replacements[$from] = $to;
  $from_html = str_replace("\u2019", "&rsquo;", $from);
  if ($from_html !== $from) {
    $replacements[$from_html] = str_replace("\u2019", "&rsquo;", $to);
  }
  $from_straight = str_replace("\u2019", "'", $from);
  if ($from_straight !== $from && !isset($replacements[$from_straight])) {
    $replacements[$from_straight] = str_replace("\u2019", "'", $to);
  }
}

  $total_potential_matches += substr_count($old_content, $from);
}

  return ['error' => 'no substitution happened', 'pairs_count' => count($pairs), 'extended_count' => count($replacements), 'total_substr_matches' => $total_potential_matches];
}
